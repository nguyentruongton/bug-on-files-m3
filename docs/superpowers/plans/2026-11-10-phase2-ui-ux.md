Wire these helpers into the Electron IPC handler and React UI components while upgrading CSS stylesheets to strict Material Design 3 tokens.

describe('resolvePlaces', () => {
  it('falls back to ~/FolderName when getSystemPath returns homeDir or throws', () => {
    const places = resolvePlaces('/home/stark', (key) => {
      if (key === 'home') return '/home/stark';
      if (key === 'documents') return '/home/stark'; // XDG fallback bug on Linux
      if (key === 'downloads') return '/home/stark/';
      if (key === 'music') throw new Error('not configured');
      return `/home/stark/${key[0].toUpperCase()}${key.slice(1)}`;
    });

    expect(places).toEqual([
      { name: 'Home', path: '/home/stark', icon: '🏠' },
      { name: 'Desktop', path: '/home/stark/Desktop', icon: '🖥️' },
      { name: 'Documents', path: '/home/stark/Documents', icon: '📄' },
      { name: 'Downloads', path: '/home/stark/Downloads', icon: '⬇️' },
      { name: 'Music', path: '/home/stark/Music', icon: '🎵' },
      { name: 'Pictures', path: '/home/stark/Pictures', icon: '🖼️' },
      { name: 'Videos', path: '/home/stark/Videos', icon: '🎥' },
    ]);
  });
});
```

describe('normalizePath and getActiveDriveMountpoint', () => {
  it('normalizes trailing slashes while preserving root /', () => {
    expect(normalizePath('/')).toBe('/');
    expect(normalizePath('/home/stark/')).toBe('/home/stark');
    expect(normalizePath('/home/stark')).toBe('/home/stark');

    expect(getActiveDriveMountpoint('/home/stark/Documents', mounts)).toBe('/home');
    expect(getActiveDriveMountpoint('/run/media/stark/USB/photos', mounts)).toBe('/run/media/stark/USB');
    expect(getActiveDriveMountpoint('/etc/nginx', mounts)).toBe('/');
    expect(getActiveDriveMountpoint('app://dashboard', mounts)).toBeNull();
  });
});
```
